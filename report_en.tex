\section{Definitions and Observables}

\subsection{Chiral Condensates}
The bare chiral condensate on a lattice volume $V = N_s^3 \times N_t$ is defined as
\begin{equation}
\langle \bar{\psi}\psi \rangle_q = \frac{1}{V} \left\langle \text{Tr}\left( D_q^{-1} \right) \right\rangle, \quad q \in \{l, s\}
\end{equation}
evaluated via stochastic $Z_2$ volume source vectors.

To eliminate the additive ultraviolet divergence proportional to $m_q / a^2$ as well as residual chiral symmetry breaking effects, the renormalized subtracted chiral condensate is defined as
\begin{equation}
\Delta_{l, s}^R(T) = \frac{1}{Z_m(\beta)} \left[ \langle\bar{\psi}\psi\rangle_l - \frac{m_l + m_{\text{res}}}{m_s + m_{\text{res}}} \langle\bar{\psi}\psi\rangle_s \right]
\label{eq:delta_ls}
\end{equation}
where $m_{\text{res}}(\beta)$ is the residual quark mass for Domain Wall Fermions, and $Z_m(\beta)$ is the quark mass renormalization factor (with scalar density renormalization $Z_S = 1/Z_m$) providing multiplicative renormalization. In physical units of $[\text{GeV}^3]$, the physical chiral condensate is given by $\langle\bar{\psi}\psi\rangle_{\text{sub}}^R(T) = a^{-3} \Delta_{l, s}^R(T) = (N_t T)^3 \Delta_{l, s}^R(T)$.

\subsection{Chiral Susceptibility}
The light-quark disconnected chiral susceptibility $\chi_{\text{disc}}$ is defined on a four-dimensional volume $V = N_s^3 \times N_t$ as
\begin{equation}
a^2\chi_{\text{disc}} = V \left[ \left\langle \left( \frac{1}{V} \text{Tr}(D_l^{-1}) \right)^2 \right\rangle - \left\langle \frac{1}{V} \text{Tr}(D_l^{-1}) \right\rangle^2 \right]
\end{equation}
The unbiased quadratic cross-product estimator for $k$ stochastic noise vectors $O_{l, i} = \text{Tr}(D_l^{-1})_i$:
\begin{equation}
\bar{O} = \frac{1}{k} \sum_{i=1}^k O_{l, i}, \quad \overline{O^2} = \frac{1}{k(k-1)} \left[ \left(\sum_{i=1}^k O_{l, i}\right)^2 - \sum_{i=1}^k O_{l, i}^2 \right]
\end{equation}
In physical units of $[\text{GeV}^2]$, the disconnected chiral susceptibility is given by
\begin{equation}
\chi_{\text{disc}}(T) = (N_t T)^2 a^2\chi_{\text{disc}} \quad [\text{GeV}^2]
\end{equation}
where $T$ is expressed in $\text{GeV}$.
